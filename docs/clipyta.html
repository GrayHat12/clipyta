176    def _formatter(self, key: Any, text: Any = ...) -> str:
177        """
178        Calculates padding and formats a given cell's text based on column width and alignment strategy.
179
180        Relies on pre-calculated maximum cell widths (`__longest_cell_lengths`) to ensure uniform 
181        column dimensions across the rendered table.
182
183        Args:
184            key (Any): The column key or header, used to look up the maximum required width for this specific column.
185            text (Any, optional): The actual text payload to display inside the cell. If set to Ellipsis (...), 
186                the method defaults to using the `key` value as the display text. Defaults to Ellipsis.
187
188        Returns:
189            str: The fully padded, aligned string bounded by an extra margin spacing, ready for standard output.
190
191        Example:
192            >>> # Assuming the maximum recorded width for the 'Status' column is 6
193            >>> renderer._formatter('Status', 'OK')
194            '   OK   '
195        """
196        if text is Ellipsis:
197            text = key
198        if not isinstance(key, str):
199            key = str(key)
200        if not isinstance(text, str):
201            text = str(text)
202
203        return self.__align.align(text)(self.__longest_cell_lengths[key] + 2)

229    def render(self, data: list[dict[Any, Any]]) -> None:
230        """
231        Processes, flattens, formats, and outputs a list of dictionaries as an ASCII table.
232
233        This method handles the entire execution pipeline: it flattens nested dictionaries/lists, applies 
234        key/value transformations, computes maximum necessary column dimensions, and dynamically draws the 
235        table layout. If called consecutively, it utilizes ANSI escape sequences to overwrite the previously 
236        drawn table output, creating an in-place updating effect.
237
238        Args:
239            data (list[dict[Any, Any]]): The raw dataset to be rendered. Elements can contain nested 
240                iterables or dictionaries.
241
242        Returns:
243            None: The final formatted table is printed directly to the standard output.
244
245        Example:
246            >>> renderer = genrerate_renderer()
247            >>> data = [{'user': {'name': 'Alice', 'role': 'Admin'}}]
248            >>> renderer.render(data)
249            # Output is visually drawn to the console, featuring 'user.name' and 'user.role' columns.
250        """
251        if isinstance(self.__row_sorting, SortConfig):
252            data = sorted(data, key=self.__row_sorting.sort,
253                          reverse=self.__row_sorting.reverse)
254        for idx in range(len(data)):
255            data[idx] = flatten(data[idx])
256            for key in list(data[idx].keys()):
257                transformation = self.__transformations.get(key)
258                if transformation:
259                    value = data[idx].pop(key)
260                    if transformation['value'] is not None:
261                        value = transformation['value'](value)
262                    if transformation['name'] is not None:
263                        key = transformation['name']
264                    data[idx][key] = value
265                self.__all_keys.add(key if isinstance(key, str) else str(key))
266
267        for key in self.__all_keys:
268            self.__longest_cell_lengths[key] = max(
269                len(str(key)), self.__longest_cell_lengths[key])
270        for item in data:
271            for key, value in item.items():
272                self.__longest_cell_lengths[key] = max(
273                    len(str(value)), self.__longest_cell_lengths[key])
274
275        all_keys_lst = self.__sort_keys(tuple(self.__all_keys))
276
277        header = self.__border_config['vertical_bold'] + self.__border_config['vertical_bold'].join(
278            map(self._formatter, all_keys_lst)) + self.__border_config['vertical_bold']
279
280        total_width = len(header)
281
282        horizontal_column_prim = self.__border_config['horizontal_bold'] * total_width
283        horizontal_column_seco = self.__border_config['horizontal'] * total_width
284
285        if self.__lines > 0:
286            print('\033[F'*self.__lines + '\033[K', end='')
287            self.__lines = 0
288
289        print(horizontal_column_prim)
290        print(header)
291        print(horizontal_column_prim)
292
293        self.__lines += 3
294
295        for item in data:
296            out = self.__border_config['vertical']
297            for key in all_keys_lst:
298                value = item.get(key, '-')
299                out += self._formatter(key, value)
300                out += self.__border_config['vertical']
301            print(out)
302            print(horizontal_column_seco)
303            self.__lines += 2

Attributes:
  • name (str | None): The new string to use as the column header. If None, the original key is kept.
  • value (Callable[[Any], Any] | None): A function that takes the original cell value and returns the formatted value. If None, the original value is kept.
Example:
>>> status_transform: Transformation = {
...     "name": "Current Status",
...     "value": lambda v: str(v).upper()
... }
name: str | None
value: Callable[[Any], Any] | None
